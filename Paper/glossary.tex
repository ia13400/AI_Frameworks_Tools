% Glossary entries

\newglossaryentry{ai}{
    name={AI},
    first={Artificial Intelligence (AI)},
    description={Field of computer science focused on developing systems capable of performing tasks that normally require human intelligence, such as learning, reasoning, problem solving, perception, and language understanding}
}

\newglossaryentry{llm}{
    name={LLM},
    first={Large Language Model (LLM)},
    description={A type of artificial intelligence model trained on large-scale text data to understand, generate, and analyze natural language}
}

\newglossaryentry{nlp}{
    name={NLP},
    first={Natural Language Processing (NLP)},
    description={Field of artificial intelligence concerned with enabling computers to process, analyze, understand, and generate human language}
}

\newglossaryentry{transformer}{
    name={Transformer},
    description={Neural network architecture based on self-attention mechanisms, widely used in modern language models}
}

\newglossaryentry{attention}{
    name={Attention Mechanism},
    description={Technique that allows a neural network to assign different levels of importance to different input tokens}
}

\newglossaryentry{attentionhead}{
    name={Attention Head},
    description={Component of a Transformer layer that learns specific relationships between tokens using the attention mechanism}
}

\newglossaryentry{mechanisticinterpretability}{
    name={Mechanistic Interpretability},
    description={Research field that studies the internal computations, representations, and mechanisms of neural networks in order to understand how they produce outputs}
}

\newglossaryentry{sentimentanalysis}{
    name={Sentiment Analysis},
    description={Natural Language Processing task that aims to identify or classify the emotional tone, opinion, or attitude expressed in a text}
}

\newglossaryentry{token}{
    name={Token},
    description={Basic unit of text processed by a language model, such as a word, subword, character, or symbol}
}

\newglossaryentry{activation}{
    name={Activation},
    description={Numerical output of a neuron or layer in a neural network after processing an input}
}
