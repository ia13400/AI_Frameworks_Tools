% =========================================================
% Glossareinträge
% =========================================================

\newglossaryentry{ai}{
    name={KI},
    first={Künstliche Intelligenz (KI)},
    description={
        Teilgebiet der Informatik, das sich mit der Entwicklung von Systemen beschäftigt,
        die Aufgaben ausführen können, die normalerweise menschliche Intelligenz erfordern,
        beispielsweise Lernen, Schlussfolgern, Problemlösen, Wahrnehmung und Sprachverstehen
    }
}

\newglossaryentry{llm}{
    name={LLM},
    first={Large Language Model (LLM)},
    description={
        Ein großes Sprachmodell, das auf umfangreichen Textdaten trainiert wird, um natürliche Sprache
        zu verstehen, zu generieren und zu analysieren
    }
}

\newglossaryentry{nlp}{
    name={NLP},
    first={Natural Language Processing (NLP)},
    description={
        Teilgebiet der Künstlichen Intelligenz, das sich mit der Verarbeitung, Analyse,
        dem Verständnis und der Generierung menschlicher Sprache durch Computer beschäftigt
    }
}

\newglossaryentry{transformer}{
    name={Transformer},
    description={
        Neuronale Netzwerkarchitektur, die auf Self-Attention-Mechanismen basiert und eine zentrale
        Grundlage moderner Sprachmodelle bildet
    }
}

\newglossaryentry{attention}{
    name={Attention-Mechanismus},
    description={
        Verfahren, das es einem neuronalen Netz ermöglicht, unterschiedliche Bestandteile einer
        Eingabesequenz unterschiedlich stark zu gewichten
    }
}

\newglossaryentry{attentionhead}{
    name={Attention Head},
    plural={Attention Heads},
    description={
        Einzelne Komponente innerhalb einer Transformer-Schicht, die bestimmte Beziehungen zwischen
        Tokens mithilfe des Attention-Mechanismus lernt
    }
}

\newglossaryentry{mechanisticinterpretability}{
    name={Mechanistische Interpretierbarkeit},
    description={
        Forschungsgebiet, das die internen Berechnungen, Repräsentationen und Mechanismen neuronaler
        Netze untersucht, um besser zu verstehen, wie diese Modelle ihre Ausgaben erzeugen
    }
}

\newglossaryentry{sentimentanalysis}{
    name={Sentimentanalyse},
    description={
        Aufgabe der Verarbeitung natürlicher Sprache, bei der die emotionale Tonalität,
        Meinung oder Haltung in einem Text erkannt oder klassifiziert wird
    }
}

\newglossaryentry{token}{
    name={Token},
    plural={Tokens},
    description={
        Grundeinheit eines Textes, die von einem Sprachmodell verarbeitet wird, beispielsweise ein Wort,
        ein Wortbestandteil, ein Zeichen oder ein Symbol
    }
}

\newglossaryentry{activation}{
    name={Aktivierung},
    plural={Aktivierungen},
    description={
        Numerischer Ausgabewert eines Neurons oder einer Schicht in einem neuronalen Netz nach der
        Verarbeitung einer Eingabe
    }
}
